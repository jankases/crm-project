// 🌟 ฟังก์ชันดึงชื่อ Primary Hospital แม่นยำตามสถาปัตยกรรม DB function getHospitalNameFromDocOrVisit(docObj, visitObj) { let targetPrimaryHospId = null; if (docObj) { // 1. เช็ค Primary Direct ID จากคอลัมน์ Hospital_ID ก่อน targetPrimaryHospId = docObj.Hospital_ID || docObj.hospital_id; // 2. ถ้าในคอลัมน์ปกติไม่มี ให้แกะ Workplaces_JSON หาเฉพาะก้อนที่เป็น Primary ("isPrimary": true) if (!targetPrimaryHospId && docObj.Workplaces_JSON) { try { let workplaces = typeof docObj.Workplaces_JSON === 'string' ? JSON.parse(docObj.Workplaces_JSON) : docObj.Workplaces_JSON; if (Array.isArray(workplaces) && workplaces.length > 0) { // 🎯 เจาะจงดึงเฉพาะตัวที่เป็น Primary เท่านั้น let primaryItem = workplaces.find(w => w.isPrimary === true || w.isPrimary === 'true' || w.type === 'Primary' || w.Type === 'Primary'); // Safety Fallback: ถ้าลืมติ๊ก isPrimary ให้หยิบโรงพยาบาลแรกสุด if (!primaryItem) primaryItem = workplaces[0]; targetPrimaryHospId = primaryItem.hospitalId || primaryItem.Hospital_ID || primaryItem.hospital_id; } } catch (e) { console.error("Parse Workplaces_JSON error:", e); } } // นำ targetPrimaryHospId ไปแมปหาชื่อโรงพยาบาลจาก globalAllHospitals if (targetPrimaryHospId) { const hospObj = globalAllHospitals.find(h => String(h.Hospital_ID || h.id || h.Hospital).trim().toLowerCase() === String(targetPrimaryHospId).trim().toLowerCase() ); if (hospObj) return hospObj.Known_As || hospObj.Hospital; // ถ้าเป็นข้อความชื่อโรงพยาบาลอยู่แล้ว ไม่ใช่ UUID ให้คืนค่าออกไปได้เลย if (typeof targetPrimaryHospId === 'string' && targetPrimaryHospId.length > 2 && !targetPrimaryHospId.includes('-')) { return targetPrimaryHospId; } } // 3. ค้นหาความสัมพันธ์ Primary ในตาราง Assignment if (window.VisitManagerCache && window.VisitManagerCache.assignments) { const docIdStr = String(docObj.Doc_ID || docObj.doc_id || docObj.id); const matchedAssign = window.VisitManagerCache.assignments.find(a => String(a.Account_ID || a.Doc_ID) === docIdStr && (a.Type === 'Doctor' || a.Type === 'Workplace') && (a.Is_Primary === true || a.IsPrimary === true || a.is_primary === true || true) ); if (matchedAssign) { const assignHospId = matchedAssign.Hospital_ID || matchedAssign.Workplace_ID; const hospObj = globalAllHospitals.find(h => String(h.Hospital_ID || h.id) === String(assignHospId)); if (hospObj) return hospObj.Known_As || hospObj.Hospital; } } } // 4. Fallback: ดึงชื่อโรงพยาบาลที่ระบุไว้ใน Visit Log นั้นๆ if (visitObj) { const directHosp = visitObj.Hospital || visitObj.Hospital_Name || visitObj.hospital; if (directHosp && String(directHosp).trim() !== '' && directHosp !== '-') return directHosp; } return '-'; }
